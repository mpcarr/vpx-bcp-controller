cmake_minimum_required(VERSION 3.24)
project(VPXBCP VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(VPX_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../vpinball" CACHE PATH "VPX source checkout")
if(NOT EXISTS "${VPX_SOURCE_DIR}/plugins/plugins/ScriptablePlugin.h")
  message(FATAL_ERROR "Set VPX_SOURCE_DIR to a vpinball checkout containing the plugin SDK")
endif()
find_package(Threads REQUIRED)
add_library(bcp-core STATIC src/BcpClient.cpp src/Message.cpp src/Launch.cpp)
set_target_properties(bcp-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(bcp-core PUBLIC src)
target_link_libraries(bcp-core PUBLIC Threads::Threads)
if(WIN32)
  target_compile_definitions(bcp-core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(bcp-core PUBLIC ws2_32 shell32)
endif()
if(APPLE)
  target_link_libraries(bcp-core PUBLIC "-framework CoreFoundation")
endif()
add_library(BCPPlugin MODULE src/Plugin.cpp)
target_include_directories(BCPPlugin PRIVATE "${VPX_SOURCE_DIR}/plugins")
target_link_libraries(BCPPlugin PRIVATE bcp-core)
set_target_properties(BCPPlugin PROPERTIES PREFIX "" OUTPUT_NAME "plugin-bcp")
if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 8)
  set_target_properties(BCPPlugin PROPERTIES OUTPUT_NAME "plugin-bcp64")
elseif(APPLE)
  set_target_properties(BCPPlugin PROPERTIES SUFFIX ".dylib")
endif()
set_target_properties(BCPPlugin PROPERTIES
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/bcp"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/bcp")
add_custom_command(TARGET BCPPlugin POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/plugin.cfg" "$<TARGET_FILE_DIR:BCPPlugin>/plugin.cfg")
install(TARGETS BCPPlugin LIBRARY DESTINATION plugins/bcp RUNTIME DESTINATION plugins/bcp)
install(FILES plugin.cfg DESTINATION plugins/bcp)
include(CTest)
if(BUILD_TESTING)
  add_executable(bcp-tests tests/tests.cpp)
  target_link_libraries(bcp-tests PRIVATE bcp-core)
  add_test(NAME protocol COMMAND bcp-tests)
  add_test(NAME launch COMMAND bcp-tests --launch-test)
  add_test(NAME executable-resolution COMMAND bcp-tests --resolve-test)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME transport COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/transport.py" $<TARGET_FILE:bcp-tests>)
  add_executable(binding-tests tests/bindings.cpp src/Plugin.cpp)
  target_include_directories(binding-tests PRIVATE "${VPX_SOURCE_DIR}/plugins")
  target_link_libraries(binding-tests PRIVATE bcp-core)
  add_test(NAME bindings COMMAND binding-tests)
  add_test(NAME binding-transport COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/transport.py" $<TARGET_FILE:binding-tests>)
endif()
